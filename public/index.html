    ⚡ CI/CD & CLI Quickstart
    
          
    
          
    Review local repos or CI pull requests with one command:
    
          
    # 1. Add and review a local repository
npx openrabbit repo add /path/to/my-repo
npx openrabbit repo commits /path/to/my-repo
npx openrabbit review --repo-path /path/to/my-repo --commit <sha>

# 2. Review a remote GitHub PR on demand
npx openrabbit review --github owner/repo#123

# 3. Review a remote GitLab MR on demand
npx openrabbit review --gitlab project/id#45

# 4. Review a remote Bitbucket PR on demand
npx openrabbit review --bitbucket workspace/repo#78
    

          
    .github/workflows/openrabbit.yml
    
          
    name: OpenRabbit AI PR Review
on:
  pull_request:
    types: [opened, synchronize, reopened]

jobs:
  openrabbit-review:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - run: npm ci
      - run: npx openrabbit review --base origin/${{ github.base_ref }} --post-github ${{ github.repository }}#${{ github.event.pull_request.number }}
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
